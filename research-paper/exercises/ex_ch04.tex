\section*{Exercises for Chapter~\ref{ch:congruences}}
\addcontentsline{toc}{section}{Exercises for Chapter 4}

\begin{enumerate}
  \item Find the remainder when $7^{100}$ is divided by $13$.
  \item Find the last two digits of $3^{200}$.
  \item Find the inverse of $17$ modulo $3120$, using \texttt{ext\_gcd}. Check your answer by hand.
  \item Solve the linear congruence $9x\equiv 6\pmod{15}$. Explain why there are several solutions and list them.
  \item Compute $\varphi(n)$ for $n=36,\ 60$ and $100$, both by hand and with \texttt{phi}.
  \item Prove that $\varphi(n)$ is even for every $n\ge 3$.
  \item Prove that $n^{561}\equiv n\pmod{561}$ for every integer $n$. (Note: $561=3\cdot 11\cdot 17$ is a Carmichael number, so Fermat's theorem alone does not show it is prime.)
  \item Show that if $\gcd(a,n)=1$, then $a^{k}\equiv a^{k'}\pmod n$ whenever $k\equiv k'\pmod{\varphi(n)}$.
  \item Show that $1\cdot 2\cdots (p-1)\equiv -1\pmod p$ for every prime $p$ (Wilson's theorem). Check it for $p=5,7,11$.
  \item Explain why the Fermat test for primality, which checks whether $a^{n-1}\equiv 1\pmod n$, can be fooled by a Carmichael number.
\end{enumerate}
