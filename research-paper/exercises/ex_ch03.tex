\section*{Exercises for Chapter~\ref{ch:primes}}
\addcontentsline{toc}{section}{Exercises for Chapter 3}

\begin{enumerate}
  \item Write $360$, $1000$ and $2310$ as products of primes.
  \item Use the fundamental theorem of arithmetic to find $\gcd(2^4\cdot 3^2\cdot 5,\ 2^2\cdot 3\cdot 7^2)$.
  \item Show that if $n=2^a 3^b$, then the number of positive divisors of $n$ is $(a+1)(b+1)$. Generalise to any product of distinct primes.
  \item Prove that $\sqrt{2}$ is irrational, using unique factorisation.
  \item Prove that there are infinitely many primes of the form $4k+3$. (Hint: consider $4p_1\cdots p_k-1$ for primes $p_i$ of this form, with $p_i$ odd.)
  \item Use the sieve of Eratosthenes by hand to list the primes up to $50$. Confirm your answer against the output of \texttt{sieve(50)}.
  \item Show that every prime $p>3$ satisfies $p^2\equiv 1\pmod{24}$.
  \item Prove that a composite integer $n$ has a prime divisor at most $\sqrt{n}$ (Lemma~\ref{lem:sqrt}), and use this to show that $91$ is composite.
  \item Show that if $n\ge 2$ and $n!+1$ is composed of primes greater than $n$, then Euclid's argument gives a new prime. Compute $2!+1$, $3!+1$ and $4!+1$.
  \item Prove that the product of any $k$ consecutive integers is divisible by $k!$.
\end{enumerate}
