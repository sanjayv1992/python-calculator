\section*{Exercises for Chapter~\ref{ch:primroots}}
\addcontentsline{toc}{section}{Exercises for Chapter 5}

\begin{enumerate}
  \item Find the order of $2$ modulo $11$, $13$ and $17$.
  \item Find a primitive root modulo $11$, and list all its powers modulo $11$.
  \item Show that $\operatorname{ord}_n(a^k)=\operatorname{ord}_n(a)/\gcd(k,\operatorname{ord}_n(a))$.
  \item Show that modulo $8$ no element has order $4$. Deduce that $\mathbb{Z}_8^{\times}$ is not cyclic.
  \item Compute the discrete logarithm of $8$ to base $2$ modulo $11$.
  \item Use Euler's criterion (Theorem~\ref{thm:euler-criterion}) to decide whether $2$ is a quadratic residue modulo $17$, $19$ and $23$.
  \item Prove that $-1$ is a quadratic residue modulo $p$ if and only if $p\equiv 1\pmod 4$, for odd primes $p$.
  \item List the quadratic residues modulo $13$ and modulo $19$.
  \item Using Proposition~\ref{prop:legendre-mult}, show that the product of two non-residues is a residue.
  \item Let $p$ be an odd prime. Show that exactly $(p-1)/2$ nonzero residues are quadratic residues.
\end{enumerate}
